    async function createSnapshotHandler() {
      setSnapCreating(true);
      setSnapError(null);
      try {
        const meta = await api.createSnapshot(snapLabel.trim() || undefined);
        setSnaps((prev) => [meta, ...prev]);
        setSnapLabel("");

      } finally {
        setSnapCreating(false);
      }
    }
