```python
from __future__ import annotations
import logging
import pytest
from pydantic import ValidationError
from app.config import Settings

_DEFAULT_TOKEN = "change-me-please"


def test_empty_token_rejected():
    with pytest.raises(ValidationError, match="must not be empty"):
        Settings(kompass_token="", database_url="sqlite:///./test.db")


def test_default_token_is_accepted():
    s = Settings(kompass_token=_DEFAULT_TOKEN, database_url="sqlite:///./test.db")
    assert s.kompass_token == _DEFAULT_TOKEN


def test_default_token_warning(caplog):
    import asyncio
    from contextlib import asynccontextmanager
    from app.main import lifespan, app as fastapi_app

    with caplog.at_level(logging.WARNING, logger="app.main"):
        asyncio.get_event_loop().run_until_complete(
            _run_lifespan(fastapi_app)
        )
    assert any("change-me-please" in r.message for r in caplog.records if r.levelno >= logging.WARNING)


async def _run_lifespan(app):
    from app.main import lifespan
    async with lifespan(app):
        pass
```
